cmake_minimum_required(VERSION 3.16)
project(modelstack_tensorbench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(tensorbench native/tensorbench.cpp)
if(MSVC)
    target_compile_options(tensorbench PRIVATE /W4 /EHsc)
else()
    target_compile_options(tensorbench PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME tensorbench_check COMMAND tensorbench --check --size 16 --iterations 3)
add_test(NAME tensorbench_small COMMAND tensorbench --check --size 1 --iterations 2)
